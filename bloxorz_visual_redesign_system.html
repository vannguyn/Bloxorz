Closed Bridge vs Heavy Switch

In the original flaw, the closed bridge rendered an intense red 'X', directly mirroring the Heavy Switch icon. In the redesign, the red X is entirely removed in favor of a ghosted wireframe boundary on the void abyss floor.

Current Flawed Behavior

Players mistake the red X closed bridge as a stepping tile or an interactive trigger button, triggering frustrating fall deaths.

Redesigned Affordance

Void abyss depth is maintained. Subtle corner brackets / wireframe indicate future bridge docking without false walkable affordance.

Renderer Technical Directive:

Draw wireframe polygon at Y + tile_depth with color (40, 80, 110) and alpha=90. Do NOT fill tile top.

1. board_renderer.py — Full-Coverage Fragile Hazard Stripes 
import pygame

def draw_fragile_cell_2d(surface: pygame.Surface, rect: pygame.Rect, stripe_spacing: int = 8):
    """
    Renders 100% full-coverage 45-degree diagonal hazard stripes across the entire
    fragile tile rect using pygame.Rect.clipline to avoid the half-tile cutoff bug.
    """
    # Base orange fill & border
    COLOR_FRAGILE_BG = (220, 115, 30)
    COLOR_FRAGILE_STRIPE = (255, 185, 45)
    COLOR_FRAGILE_BORDER = (160, 75, 15)

    pygame.draw.rect(surface, COLOR_FRAGILE_BG, rect)

    # Offset must span from 0 to width + height so diagonal lines cover all corners
    max_span = rect.width + rect.height
    for offset in range(0, max_span, stripe_spacing):
        start_pt = (rect.left + offset, rect.top)
        end_pt = (rect.left, rect.top + offset)

        # Clip line to strictly stay within tile bounding rect
        clipped = rect.clipline(start_pt, end_pt)
        if clipped:
            pygame.draw.line(surface, COLOR_FRAGILE_STRIPE, clipped[0], clipped[1], width=2)

    # Crisp perimeter warning border
    pygame.draw.rect(surface, COLOR_FRAGILE_BORDER, rect, width=2)
2. isometric_board_renderer.py — Extruded Tactile 3D Switches 
import pygame

def draw_switch_3d(surface: pygame.Surface, center_x: int, center_y: int, 
                    switch_type: str, is_pressed: bool):
    """
    Renders switches with 3D physical height (pedestal + extruded cap)
    and depresses along the Y-axis when stepped on by the player's block.
    """
    # Cap elevation relative to tile face: -5px when relaxed, -1px when pressed
    lift_y = -1 if is_pressed else -5
    cap_y = center_y + lift_y

    if switch_type == 'SOFT':  # Circular Amber Button
        radius_x, radius_y = 12, 6
        # 1. Base ring / shadow bevel
        pygame.draw.ellipse(surface, (70, 50, 10), (center_x - radius_x, center_y - radius_y + 2, radius_x*2, radius_y*2))
        # 2. Cylindrical body skirt
        if not is_pressed:
            pygame.draw.rect(surface, (160, 120, 20), (center_x - radius_x, cap_y, radius_x*2, 4))
        # 3. Elevated top button face
        top_color = (255, 230, 80) if is_pressed else (240, 190, 30)
        pygame.draw.ellipse(surface, top_color, (center_x - radius_x, cap_y - radius_y, radius_x*2, radius_y*2))
        pygame.draw.ellipse(surface, (255, 255, 180), (center_x - radius_x, cap_y - radius_y, radius_x*2, radius_y*2), width=1)

    elif switch_type == 'HEAVY':  # Heavy Industrial Crimson X
        # 1. Cast metal base frame (Octagon or small sub-plate)
        base_pts = [
            (center_x, center_y - 10), (center_x + 16, center_y),
            (center_x, center_y + 10), (center_x - 16, center_y)
        ]
        pygame.draw.polygon(surface, (45, 50, 60), base_pts)
        pygame.draw.polygon(surface, (75, 85, 100), base_pts, width=1)

        # 2. Extruded 3D 'X' crossbars
        arm_w = 4
        # Lower depth shadow for X arms
        if not is_pressed:
            pygame.draw.line(surface, (110, 15, 25), (center_x - 10, center_y - 5), (center_x + 10, center_y + 5), width=arm_w + 1)
            pygame.draw.line(surface, (110, 15, 25), (center_x - 10, center_y + 5), (center_x + 10, center_y - 5), width=arm_w + 1)
        # Upper face X arms
        cross_color = (255, 60, 70) if is_pressed else (225, 30, 45)
        pygame.draw.line(surface, cross_color, (center_x - 10, cap_y - 5), (center_x + 10, cap_y + 5), width=arm_w)
        pygame.draw.line(surface, cross_color, (center_x - 10, cap_y + 5), (center_x + 10, cap_y - 5), width=arm_w)
3. isometric_board_renderer.py — Optical Depth Goal Abyss Shaft 
import pygame

def draw_goal_abyss_3d(surface: pygame.Surface, top_corners: list[tuple[int, int]], shaft_depth: int = 32):
    """
    Renders an authentic deep optical pit into the void. Instead of a raised golden ring,
    this projects inward walls that converge into pitch-black abyss with interior shading.
    corners: [Top, Right, Bottom, Left]
    """
    T, R, B, L = top_corners
    center_x = (T[0] + B[0]) // 2
    center_y = (T[1] + B[1]) // 2

    # Scaled down bottom aperture sunken deep along Y-axis
    scale = 0.55
    inner_corners = [
        (center_x + int((pt[0] - center_x) * scale), center_y + int((pt[1] - center_y) * scale) + shaft_depth)
        for pt in [T, R, B, L]
    ]
    iT, iR, iB, iL = inner_corners

    # 1. Left internal shaft wall (shaded dark)
    pygame.draw.polygon(surface, (18, 22, 32), [L, B, iB, iL])
    # 2. Right internal shaft wall (shaded mid-dark)
    pygame.draw.polygon(surface, (28, 34, 48), [B, R, iR, iB])
    # 3. Bottom abyss floor (pure space void)
    pygame.draw.polygon(surface, (6, 8, 12), [iT, iR, iB, iL])

    # 4. Subtle gold chamfer edge on top perimeter to attract the eye
    pygame.draw.polygon(surface, (190, 155, 50), top_corners, width=2)
4. isometric_board_renderer.py — Closed Bridge Ghost Wireframe 
import pygame

def draw_closed_bridge_isometric(surface: pygame.Surface, top_corners: list[tuple[int, int]], tile_depth: int):
    """
    Replaces the misleading Red-X on closed bridges. Renders an empty void space
    with a faint recessed docking rail at the bottom abyss level (Y + tile_depth).
    """
    # Shift corners down to bottom depth level
    bottom_corners = [(x, y + tile_depth) for (x, y) in top_corners]
    
    # 1. Subtle dotted or faint docking outline at pit depth (no solid floor!)
    dock_wireframe_color = (35, 60, 80)
    pygame.draw.polygon(surface, dock_wireframe_color, bottom_corners, width=1)

    # 2. Corner docking brackets at normal surface height to show where the bridge lands
    bracket_color = (45, 85, 110)
    for (cx, cy) in top_corners:
        pygame.draw.circle(surface, bracket_color, (cx, cy), radius=2)
